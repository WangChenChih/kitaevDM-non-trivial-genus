\begin{abstract}
    The reduced density matrix of a multiply connected subsystem exhibits an exact diagonal structure different from that of a simply connected subsystem. 
\end{abstract}

First, we consider a non-simply connected region $A$ with an interior hole $B$. The reduced density matrix of this subsystem is as follows,
\[
    \rho_{A}
    =
    \frac{1}{2^{N^A - N^A_p}}
    \hat{\Pi}_A
    \lr{\mathds{1} + W_\mathcal{L}}
    \sum_{[\mathfrak{D}]\subset A}
    \lr{
        \mathfrak{C}_\mathfrak{D} \Sigma_\mathfrak{D} 
        +
        \mathfrak{C}_{\mathfrak{D} \ominus \mathfrak{H}} \Sigma_{\mathfrak{D}\ominus\mathfrak{H}}
    }
\]
where $\mathcal{L}$ is a closed loop inside the subsystem $A$ that encloses the hole $B$. In other words, the loop $\mathcal{L}$ is non-contractible in the non-simply connected subsystem $A$. The edge configuration $\mathfrak{H}$ represents the \textit{patch} configuration of the hole enclosed by $A$ as illustrated in Fig.~\ref{fig:patch-config}(a). Note that $\Sigma_{\mathfrak{D}\ominus \mathfrak{H}}$ is shorthand notation for the string operator of the combination of $\mathfrak{D}$ combined with the patch configuration, but with the local identity operators generated by the relation $\sigma^x\sigma^y\sigma^z\propto\mathds{1}$ inside the hole $B$ excluded.

    \rho_A

        \Sigma_\mathfrak{H}
    }

\]

    \frac{1}{2^{N^A - N^A_p - 2}}

    \lrm{
        p_\mathfrak{D}^{(+)}
        \lr{

        }

        \lr{

        }

\]

    \rho_A
    \equiv
    \frac{1}{2^{N^A - N^A_p - 2}}

    \hat{\Pi}_\mathcal{L}
    \sum_{[\mathfrak{D}]\subset A}
    \lrm{
        \lr{
        \hat{\Pi}_\mathfrak{H}^{(+)}
        \tilde{\mathfrak{C}}^{(+)}_\mathfrak{D} \Sigma_\mathfrak{D}
        }
        \oplus
        \lr{
        \hat{\Pi}_\mathfrak{H}^{(-)}

        }
    }
\]

\left\{

        =

                1
                +

            } \\
    & p_\mathfrak{D}^{(-)} = 1 - p_\mathfrak{D}^{(+)}
\end{aligned}
\right.
\]

\tilde{\mathfrak{C}}^{(\pm)}_\mathfrak{D} \equiv p^{(\pm)}_\mathfrak{D} \mathfrak{C}_\mathfrak{D}
\]

\mathfrak{C}_\mathfrak{D}

\left\{

    & \tilde{\mathfrak{C}}_\varnothing^{(-)} = 1 - p_\varnothing = \frac{1 - \mathfrak{C}_\mathfrak{H}}{2}
\end{aligned}

\hat{\Pi}^{(\pm)}_\mathfrak{H}

\]

\Tr_A\lr{\hat{\Pi}_A \hat{\Pi}_\mathcal{L} \hat{\Pi}_\mathfrak{H}^{(\pm)}} = 2^{N^A - N^A_p - 2}

\begin{equation}
    \rho_A
    =
    \frac{1}{2^{N^A - N^A_p - 2g}}
    \hat{\Pi}_A
    \hat{\Pi}_g
    \sum_{[\mathfrak{D}]\subset A}
    \bigoplus_{\lrg{s}}
    \tilde{\mathfrak{C}}^{(\lrg{s})}_\mathfrak{D}
    \hat{\Pi}_\mathfrak{H}^{(\lrg{s})}
    \Sigma_\mathfrak{D}
\end{equation}
\begin{equation}
    \left\{
    \begin{aligned}
        &\hat{\Pi}_g \equiv
        \prod_{n=1}^{g}\lr{\frac{\mathds{1} + W_{\mathcal{L}^{(n)}}}{2}} \\
        &\hat{\Pi}_\mathfrak{H}^{(\lrg{s})} \equiv 
        \prod_{n=1}^{g}\lr{\frac{\mathds{1} + s_n\Sigma_{\mathfrak{H}^{(n)}}}{2}}  \\
        & \tilde{\mathfrak{C}}_\mathfrak{D}^{(\lrg{s})} \equiv \lr{\prod_{n=1}^{g}p_{\mathfrak{D},n}^{(s_n)}} \mathfrak{C}_\mathfrak{D}
    \end{aligned}
    \right.
\end{equation}
The non-contractible Wilson operators connecting two electric-field operators fall into two types. One type, denoted as $\mathfrak{W}_s$, has both endpoints attached to the same boundary, while the other, denoted as $\mathfrak{W}_d$, has endpoints attached to different boundaries. As illustrated in Fig.~\ref{fig:patch-config}(b), the non-contractible Wilson operator $W_1$ is of type $\mathfrak{W}_s$, while the other Wilson operator $W_2$ is of type $\mathfrak{W}_d$. 
\[
W_2 \hat{\Pi}^{(\pm)}_\mathfrak{H} W_2^\dagger = \hat{\Pi}^{(\mp)}_\mathfrak{H}
\]

\begin{equation}\label{eq:patition-condition}
  \sum_{\lrg{s}} \tilde{\mathfrak{C}}_\mathfrak{D}^{(\lrg{s})} 
  = \sum_{\lrg{s}} \lr{\prod_{n=1}^{g}p_{\mathfrak{D},n}^{(s_n)}} \mathfrak{C}_\mathfrak{D}
  =
  \lr{\prod_{n=1}^{g}\sum_{s_n}p_{\mathfrak{D},n}^{(s_n)}} \mathfrak{C}_\mathfrak{D}
  = \mathfrak{C}_\mathfrak{D}
\end{equation}

\begin{equation}\label{eq:S-rep_multi-connect}
    \hat{\mathcal{S}}_\mathfrak{D} 
    \equiv 
    \hat{\Pi}_A \hat{\Pi}_g
    \bigoplus_{\lrg{s}} \tilde{p}_\mathfrak{D}^{(\lrg{s})} \hat{\Pi}_\mathfrak{H}^{(\lrg{s})}\Sigma_\mathfrak{D}
\end{equation}
Since $[\Sigma_\mathfrak{H}, \Sigma_{\mathfrak{H}'}] = 0$ and $[\Sigma_\mathfrak{H}, \Sigma_\mathfrak{D}] = 0\;\forall \mathfrak{D}\subset A$, the projective string operators also satisfy the following algebraic relation
\begin{equation}\label{eq:fermion-statistics-algebra}
    \hat{\mathcal{S}}_\mathfrak{D} \hat{\mathcal{S}}_{\mathfrak{D}'} = \varphi(\mathfrak{D},\mathfrak{D}') \hat{\mathcal{S}}_{\mathfrak{D} \ominus \mathfrak{D}'}
\end{equation}
The fact that this relation still applies implies that the endpoints of the projective string operators defined in non-simply connected subsystems still obey fermionic statistics. Therefore, \textbf{the string operators are still equivalent to Majorana fermionic operators in non-simply connected subsystems}. 

Together, the reduced density matrix can be organized into the following form that resembles the Majorana fermionic reduced density matrix of the equivalent Majorana fermionic system. 
\begin{equation}\label{eq:RDM-norm}
    \rho_A
    =
    \frac{1}{2^{\abs{\de A}/2 - N^A_\de}}
    \lrm{
        \frac{1}{2^{N^A/2}}
        \sum_{[\mathfrak{D}]\subset A}
        \mathfrak{C}_\mathfrak{D} \hat{\mathcal{S}}_\mathfrak{D}
    }
\end{equation}
In the above equation, we have used the relation \eqref{eq:Pick_multi-con}, and $N^A_\de = 1 + g$ is the number of disconnected boundaries of the subsystem $A$.
\[
\Tr_A \hat{\mathcal{S}}_\varnothing 
= 
\sum_{\lrg{s}} \tilde{p}^{(\lrg{s})}_\varnothing \Tr_A
\lr{
\hat{\Pi}_A \hat{\Pi}_g \hat{\Pi}^{(\lrg{s})}_{\mathfrak{H}}
}
=
\sum_{\lrg{s}} \tilde{p}^{(\lrg{s})}_\varnothing 2^{N^A - N^A_p - 2g}
=
2^{N^A - N^A_p - 2g}
\]
On the other hand, the reduced density matrix of the equivalent Majorana fermionic system reads
\[
\rho^F_A = \frac{1}{2^{N^A/2}} \sum_{[\mathfrak{D}]\subset A} \mathfrak{C}_\mathfrak{D} \hat{\mathcal{M}}_\mathfrak{D}
\]
Accordingly, the Hilbert space dimension of the equivalent Majorana fermionic system is 
\[
\frac{\Tr_A\hat{\mathcal{S}}_\varnothing}{\Tr_{A,F}\hat{\mathcal{M}}_{\varnothing}} 
=
\frac{\dim \mathcal{H}_S}{\dim \mathcal{H}_F}
=
2^{\abs{\de A}/2 - N^A_\de}
\]
Note that. Therefore, the entanglement entropy computed using the spin reduced density matrix is
\begin{align}
    S_A
    &=
    -\Tr_A \rho_A \ln \rho_A \nonumber\\
    &= 
    -\frac{1}{2^{\abs{\de A}/2 - N^A_\de}}
    \Tr_A\lrm{
        \lr{
        \frac{1}{2^{N^A/2}}
        \sum_{[\mathfrak{D}]\subset A}
        \mathfrak{C}_\mathfrak{D} \hat{\mathcal{S}}_\mathfrak{D}
        }
        \ln\lr{
        \frac{1}{2^{N^A/2}}
        \sum_{[\mathfrak{D}]\subset A}
        \mathfrak{C}_\mathfrak{D} \hat{\mathcal{S}}_\mathfrak{D}
        }
    }
    +
    \lr{\frac{1}{2}\abs{\de A} - N^A_\de}\ln2 \nonumber\\
    &=
    -\Tr_{A,F} \rho^F_A\ln \rho^F_A
    +
    \lr{\frac{1}{2}\abs{\de A} - N^A_\de}\ln2
\end{align}

The physical meaning of the string operator $\Sigma_\mathfrak{H}$ of the patch configuration can be interpreted in another way. The string operator $\Sigma_\mathfrak{H}$ of the patch configuration can be decomposed into the multiplication of all the $\xi$ operators along the interior boundary and the Pauli spin operators that represent residual electric field inside the hole along the boundary attached to the string operators covering the hole. This decomposition is pictorially illustrated as follows. 
\begin{align}
\Sigma
\lr{\wordfig{3}{figures/web-patch.pdf}}
&=
\lr{
\wordfig{3}{figures/covering-and-xi.pdf}
} \nonumber\\
&=
\lr{\wordfig{3}{figures/xi-multiply.pdf}}
\underbrace{
\lr{\wordfig{3}{figures/covering.pdf}}
}_{\propto \mathds{1}} \nonumber\\
&=
\lr{\prod_{\hat{\xi}\in\de B}\hat{\xi}} 
\underbrace{
\lr{\prod_{\anglelr{ij}\in B}\dimer{i}{j}} \lr{\prod_{i\in\de B}\sigma^{\mu_i}_i}
}_{\propto \mathds{1} } 
\propto \prod_{\hat{\xi}\in\de B}\hat{\xi}
\end{align}
From this result, we see that the string operator $\Sigma_\mathfrak{H}$ of the patch configuration is the product of all the $\xi$-operators along the interior boundary. Accordingly, the projector $\hat{\Pi}_\mathfrak{H}$ can be redefined as follows,
\[
\hat{\Pi}_\mathfrak{H}^{(\lrg{s})} 
= 
\prod_{n=1}^{g}\frac{1}{2}\lr{\mathds{1} + s_n\prod_{\hat{\xi}\in \de B^{(n)}}\hat{\xi} }.
\]
Note that we don't have to consider this pattern in a \textbf{simply connected subsystem} since we have already included the configurations satisfying $[\mathfrak{D}\ominus\mathfrak{D}_Z]\subset A$ when constructing the reduced density matrix. The string operator defined as $\prod_{\hat{\xi}\in\de A}\hat{\xi}$ in a simply connected region $A$ is equivalent to the combination of the product of all the dimers lying outside $A$, the dimers crossing the boundary $\de A$, and the gauge links forming $\xi$-operators along the boundary. This configuration is taken into account by $[\mathfrak{D}\ominus\mathfrak{D}_Z]\subset A$.

\end{align}

\section{Generalized Geometric Relation}
This formula can be generalized to multiply-connected regions. Suppose a connected region has a genus number $g$, and the $n$-th hole surrounded by the connected region $A$ is denoted by $B^{(n)}$ such that $A\cap B^{(n)}=\varnothing$ and $A\cup \lr{\bigcup_{n=1}^{g} B^{(n)}}$ is simply-connected. Denoting by $\ell_{\mrm{out}} = \abs{\de\lr{A\cup \lr{\bigcup_{n=1}^{g} B^{(n)}}}}$ the number of crossing edges of the outer boundary, and by $\ell_{\mrm{in}}^{(n)}=\abs{\de B^{(n)}}$ the length of the $n$-th inner boundary, we have
\[
\left\{
\begin{aligned}
&\ell_{\mrm{out}} = N^A- 2N^A_{p} + \sum_{n=1}^{g}\lr{N^{B^{(n)}} -2N^{B^{(n)}}_{p} - 2N^{\de B^{(n)}}_{p}} +2 \\
&\ell_{\mrm{in}}^{(n)} = N^{B^{(n)}} -2N^{B^{(n)}}_{p}  +2
\end{aligned}
\right.
\]
where $N^{\de(n)}_p$ is the number of plaquettes at the interface between $A$ and $B$. It is obvious that $N^{\de(n)}_p = \ell_{\mrm{in}}^{(n)}$. Thus, we have
\[
\ell_{\mrm{out}} = N^A- 2N^A_{p} - \sum_{n=1}^{g}\ell_{\mrm{in}}^{(n)} -2g + 2
\]
At the same time, we know that the length of the boundary of the region $A$ is the sum of the outer boundary and all inner boundaries
\[
\abs{\de A} = \ell_{\mrm{out}} + \sum_{n=1}^{g}\ell_{\mrm{in}}^{(n)}
\]
Combining this relation with the result we obtained gives a generalized geometric relation:
\begin{equation}\label{eq:Pick_multi-con}
    \abs{\de A} = N^A-2N^A_p+2(1-g)
\end{equation}
The above relation implies that the number of lattice sites $N^A$ determines whether the total boundary length $\abs{\de A}$ is even or odd. 
\[
N^A = \abs{\de A} \mod 2
\]
